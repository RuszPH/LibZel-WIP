const grade = [];



for (i = 0; i < 8;i++){

  let num = i+1;
  let grado = parseFloat(prompt("Enter grade" + num + ": "));
  grade.push(parseFloat(grado))
}

const total = parseFloat(grade.reduce((sum,current) => sum + current,0));

let len = grade.length;
let ave = total/len;

  
console.log(ave)

if (ave > 100) {
  console.log("Invalid Grade")
}
else if (ave >= 98) {
   console.log("with Highest Honor")
  console.log("Congratulations")
}

else if (ave >= 95) {
  console.log("with High Honor")
  console.log("Congratulations")
}

else if (ave >= 90) {
  console.log("with Honor")
  console.log("Congratulations")
}

else if (ave >= 75) {
  console.log("Passed")
  console.log("Congratulations")
}

else {
  console.log("Failed")
}
